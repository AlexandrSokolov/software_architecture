#!/usr/bin/env bash
# Builds all projects in dependency order. Each project is an independent Maven build (like separate repositories).
#
#   ./build.sh                            -> clean install (default)
#   ./build.sh -Pintegration-tests        -> clean install -Pintegration-tests (only flags: default phases kept)
#   ./build.sh clean                      -> clean (explicit phases/goals replace the default)
#   ./build.sh verify -Pintegration-tests -> verify -Pintegration-tests
#
# Without install, children resolve login-tracking-parent and login-api from ~/.m2 (the last installed version).
set -euo pipefail

PROJECTS=(
  login-tracking-parent
  login-api
  login-processing-service
  track-login-service
)

DEFAULT_GOALS=(clean install)
PHASES=" clean validate compile test package verify install "

# A phase from the list or a plugin goal (prefix:goal) counts as an explicit goal.
has_goal=false
for arg in "$@"; do
  if [[ "$PHASES" == *" $arg "* || "$arg" == *:* ]]; then
    has_goal=true
    break
  fi
done

if [[ "$has_goal" == true ]]; then
  ARGS=("$@")
else
  ARGS=("${DEFAULT_GOALS[@]}" "$@")
fi

if [[ "$has_goal" == true && " ${ARGS[*]} " != *" install "* && " ${ARGS[*]} " != *" clean "* ]]; then
  echo ">>> Note: no install - children use the parent and login-api last installed in ~/.m2"
fi

ROOT_DIR="$(cd "$(dirname "$0")" && pwd)"

for project in "${PROJECTS[@]}"; do
  if [[ ! -d "$ROOT_DIR/$project" ]]; then
    echo ">>> Skipping $project (not found)"
    continue
  fi
  echo ">>> Building $project: mvn ${ARGS[*]}"
  (cd "$ROOT_DIR/$project" && ./mvnw "${ARGS[@]}")
done

echo ">>> Build finished"

#!/usr/bin/env bash
# HTTP health check for images without curl/wget (e.g. eclipse-temurin ships only bash).
# Exit code 0 if GET <path> on localhost:<port> answers HTTP 200, otherwise non-zero.
# Usage: http-check.sh <path> [port]   (port defaults to $SERVER_PORT of the container)
set -euo pipefail

path="$1"
port="${2:-$SERVER_PORT}"

# bash opens the TCP connection itself (/dev/tcp), no HTTP client needed.
exec 3<>"/dev/tcp/localhost/${port}"
printf 'GET %s HTTP/1.0\r\nHost: localhost\r\n\r\n' "${path}" >&3
head -1 <&3 | grep -q ' 200 '

# Keycloak (authorization server)

Keycloak issues access tokens for service-to-service calls.

## OAuth2 roles

| Role                 | What it is                                   | What it does                                                        |
|----------------------|----------------------------------------------|---------------------------------------------------------------------|
| Authorization server | Keycloak                                     | Authenticates clients, issues signed JWT access tokens              |
| Client               | A backend service that calls a protected API | Gets a token, sends it as `Authorization: Bearer <token>`           |
| Resource server      | The protected API                            | Validates the token locally, checks scope and audience              |

A backend service can play both roles: it is a resource server for its own callers
and a client when it calls another API.

In this project:

| Role                 | Service                    |
|----------------------|----------------------------|
| Authorization server | Keycloak                   |
| Client               | `login-processing-service` |
| Resource server      | `track-login-service`      |

## Flow (client credentials)

1. The client sends its `client_id` and `client_secret` to the token endpoint
   (`grant_type=client_credentials`: a service gets a token for itself, there is no user).
2. Keycloak returns a JWT signed with the realm's private key.
3. The client calls the resource server with `Authorization: Bearer <token>`.
4. The resource server validates the token locally: signature (public keys fetched once from
   the JWKS endpoint), `iss`, `exp`, `aud`, then the required scope.
   There is no call to Keycloak per request.

## Realm file: `realm-login-tracking.json`

Keycloak imports the file at startup (`--import-realm`), so it starts already configured.
JSON has no comments, so the attributes are explained here.

### Realm

| Attribute | Value            | Meaning                                                                      |
|-----------|------------------|------------------------------------------------------------------------------|
| `realm`   | `login-tracking` | Isolated space with its own clients and signing keys; part of the issuer URL |
| `enabled` | `true`           | Realm is active                                                              |

Issuer URL: `http://<host>:<port>/realms/<realm>`.

### Client scope (permission)

| Attribute                                  | Value                 | Meaning                                                                      |
|--------------------------------------------|-----------------------|------------------------------------------------------------------------------|
| `name`                                     | `logins:write`        | Permission name                                                              |
| `protocol`                                 | `openid-connect`      | Scope for OIDC/OAuth2 tokens                                                 |
| `include.in.token.scope`                   | `true`                | Puts the name into the `scope` claim; Spring maps it to `SCOPE_logins:write` |
| `display.on.consent.screen`                | `false`               | No user consent screen (machine-to-machine)                                  |
| `protocolMappers` → `oidc-audience-mapper` | —                     | Adds an audience to the token                                                |
| `included.custom.audience`                 | `track-login-service` | Value of the `aud` claim: the resource server the token is meant for         |
| `access.token.claim`                       | `true`                | `aud` goes into the access token                                             |
| `id.token.claim`                           | `false`               | Not needed in the ID token (no user login)                                   |

Why the audience matters: a resource server accepts only tokens with its own `aud`.
A token issued for another API is rejected (401), even if the same Keycloak signed it.
Keycloak does not set a custom `aud` by default, hence the mapper.

### Client (the calling service)

| Attribute                   | Value                             | Meaning                                                           |
|-----------------------------|-----------------------------------|-------------------------------------------------------------------|
| `clientId`                  | `login-processing-service`        | Client identifier, sent as `client_id`                            |
| `publicClient`              | `false`                           | Confidential client: it has a secret (a backend can keep it safe) |
| `clientAuthenticatorType`   | `client-secret`                   | Authenticates with `client_id` + `client_secret`                  |
| `secret`                    | `${PROCESSING_CLIENT_SECRET}`     | Placeholder, resolved from the container environment at import    |
| `serviceAccountsEnabled`    | `true`                            | Enables the client credentials flow                               |
| `standardFlowEnabled`       | `false`                           | No authorization code flow (no browser login)                     |
| `directAccessGrantsEnabled` | `false`                           | No password grant (no user credentials)                           |
| `defaultClientScopes`       | `["logins:write"]`                | Every token of this client gets the scope and the audience        |
| `optionalClientScopes`      | `[]`                              | No additional scopes on request                                   |

Principle: the client gets only what it needs (least privilege): one flow, one scope.

## Where the client secret comes from

The secret is not stored in the realm file. It is passed through three steps:

1. `.env`: `PROCESSING_CLIENT_SECRET=processing-secret-change-me`.
2. `docker-compose.yml` (keycloak service) puts it into the container environment:
   `PROCESSING_CLIENT_SECRET: ${PROCESSING_CLIENT_SECRET}`.
3. During the import, Keycloak replaces `${PROCESSING_CLIENT_SECRET}` in the realm file
   with that value.

Syntax: `${NAME}`, not `${env.NAME}`. Keycloak does not resolve `${env.NAME}` and stores the text literally:
the import succeeds, but every token request then fails with `unauthorized_client`.

The client service gets the same value from the same `.env` variable, so both sides match.
Change it in one place (`.env`) and recreate the Keycloak container (the realm is re-imported).

## Configuration (`.env`)

| Variable                   | Meaning                                                      |
|----------------------------|--------------------------------------------------------------|
| `KEYCLOAK_IMAGE`           | Pinned image version                                         |
| `KEYCLOAK_HOST_PORT`       | Port on the host (admin console, curl requests)              |
| `KEYCLOAK_ADMIN_USERNAME`  | Bootstrap admin of the admin console (realm `master`)        |
| `KEYCLOAK_ADMIN_PASSWORD`  | Bootstrap admin password                                     |
| `PROCESSING_CLIENT_SECRET` | Client secret, shared by Keycloak and the client service     |

## Local demo only

- `start-dev`: no TLS, in-memory database, the realm is re-imported at every start.
  Production: `start` with TLS, an external database, a real hostname.
- Admin `admin/admin` and the client secret in `.env` are demo values.
  Production: secrets from a secret store, never in the repository.

## Try it

Start:

    docker compose up -d keycloak

Admin console: http://localhost:8180 (log in with the admin user, switch to realm `login-tracking`).

### Where the request parameters come from

| Part of the request | Value                         | Source                                                   |
|---------------------|-------------------------------|----------------------------------------------------------|
| Host port           | `8180`                        | `.env`: `KEYCLOAK_HOST_PORT`                             |
| Realm in the URL    | `login-tracking`              | Realm file: `realm`                                      |
| `grant_type`        | `client_credentials`          | OAuth2 standard; allowed by `serviceAccountsEnabled`     |
| `client_id`         | `login-processing-service`    | Realm file: `clientId`                                   |
| `client_secret`     | `processing-secret-change-me` | `.env`: `PROCESSING_CLIENT_SECRET` (see above)           |

### 1. Discovery: what this authorization server offers

    curl -s localhost:8180/realms/login-tracking/.well-known/openid-configuration | jq

Shows the issuer and all endpoint URLs (token, JWKS, introspection, ...).
A resource server configured with `issuer-uri` reads this document at startup.

### 2. Public keys (JWKS)

    curl -s localhost:8180/realms/login-tracking/protocol/openid-connect/certs | jq

The keys used to verify token signatures. The resource server fetches them once and caches them.

### 3. Request a token (what the client service does)

    curl -s -X POST localhost:8180/realms/login-tracking/protocol/openid-connect/token \
      -d grant_type=client_credentials \
      -d client_id=login-processing-service \
      -d client_secret=processing-secret-change-me | jq

The response contains `access_token`, `expires_in` and `token_type: Bearer`.

Store the token for the next requests:

    TOKEN=$(curl -s -X POST localhost:8180/realms/login-tracking/protocol/openid-connect/token \
      -d grant_type=client_credentials \
      -d client_id=login-processing-service \
      -d client_secret=processing-secret-change-me | jq -r .access_token)

### 4. Decode the token (claims)

    echo "$TOKEN" | jq -R 'split(".")[1] | gsub("-";"+") | gsub("_";"/") | @base64d | fromjson'

A JWT is `header.payload.signature`, base64url-encoded. Expected claims:

- `"scope": "logins:write"`
- `"aud": "track-login-service"`
- `"iss": "http://localhost:8180/realms/login-tracking"`
- `"exp"`: expiry time (epoch seconds)

Note: `iss` depends on the URL used to request the token. A token requested from the host
(`localhost:8180`) has a different `iss` than one requested inside compose (`keycloak:8080`),
so a resource server configured with the compose issuer rejects it.
Use host tokens for inspection only.

### 5. Wrong secret

    curl -s -X POST localhost:8180/realms/login-tracking/protocol/openid-connect/token \
      -d grant_type=client_credentials \
      -d client_id=login-processing-service \
      -d client_secret=wrong | jq

Returns HTTP 401 with `"error": "unauthorized_client"`.

### 6. Introspection: ask Keycloak whether a token is active

    curl -s -X POST localhost:8180/realms/login-tracking/protocol/openid-connect/token/introspect \
      -d client_id=login-processing-service \
      -d client_secret=processing-secret-change-me \
      -d token="$TOKEN" | jq

Returns `"active": true` and the claims, or `"active": false` for an expired or revoked token.
This is the alternative to local JWT validation (see the tradeoff below).

### 7. Admin REST API: inspect the imported client

Admin token (realm `master`, built-in client `admin-cli`, admin user from `.env`):

    ADMIN_TOKEN=$(curl -s -X POST localhost:8180/realms/master/protocol/openid-connect/token \
      -d grant_type=password \
      -d client_id=admin-cli \
      -d username=admin \
      -d password=admin | jq -r .access_token)

Client configuration as Keycloak sees it after the import:

    curl -s -H "Authorization: Bearer $ADMIN_TOKEN" \
      "localhost:8180/admin/realms/login-tracking/clients?clientId=login-processing-service" | jq

## Token lifetime tradeoff

With local validation, a token stays valid until `exp`, even if the client is disabled
in Keycloak in the meantime. A short token lifetime (Keycloak default: 5 minutes) limits
this window. The alternative, introspection (a call to Keycloak per request), gives
immediate revocation, but adds latency and makes Keycloak a runtime dependency of every request.

# set JAVACMD and JAVACCMD
set_java_home() {
  # For Cygwin and MinGW, ensure paths are in Unix format before anything is touched
  if [ -n "${JAVA_HOME-}" ]; then
    if [ -x "$JAVA_HOME/jre/sh/java" ]; then
      # IBM's JDK on AIX uses strange locations for the executables
      JAVACMD="$JAVA_HOME/jre/sh/java"
      JAVACCMD="$JAVA_HOME/jre/sh/javac"
    else
      JAVACMD="$JAVA_HOME/bin/java"
      JAVACCMD="$JAVA_HOME/bin/javac"

      if [ ! -x "$JAVACMD" ] || [ ! -x "$JAVACCMD" ]; then
        echo "The JAVA_HOME environment variable is not defined correctly, so mvnw cannot run." >&2
        echo "JAVA_HOME is set to \"$JAVA_HOME\", but \"\$JAVA_HOME/bin/java\" or \"\$JAVA_HOME/bin/javac\" does not exist." >&2
        return 1
      fi
    fi
  else
    JAVACMD="$(
      'set' +e
      'unset' -f command 2>/dev/null
      'command' -v java
    )" || :
    JAVACCMD="$(
      'set' +e
      'unset' -f command 2>/dev/null
      'command' -v javac
    )" || :

    if [ ! -x "${JAVACMD-}" ] || [ ! -x "${JAVACCMD-}" ]; then
      echo "The java/javac command does not exist in PATH nor is JAVA_HOME set, so mvnw cannot run." >&2
      return 1
    fi
  fi
}

# hash string like Java String::hashCode
hash_string() {
  str="${1:-}" h=0
  while [ -n "$str" ]; do
    char="${str%"${str#?}"}"
    h=$(((h * 31 + $(LC_CTYPE=C printf %d "'$char")) % 4294967296))
    str="${str#?}"
  done
  printf %x\\n $h
}

  #   Trims trailing and leading whitespace, carriage returns, tabs, and linefeeds.
  #   Needed for removing poorly interpreted newline sequences when running in more

  verbose "$(ls -la "$TMP_DOWNLOAD_DIR")"

package com.tipico.demo.logintracking.api.validation;

import jakarta.validation.Constraint;
import jakarta.validation.Payload;

import java.lang.annotation.Documented;
import java.lang.annotation.Retention;
import java.lang.annotation.Target;

import static java.lang.annotation.ElementType.*;
import static java.lang.annotation.RetentionPolicy.RUNTIME;

/**
 * The value must be an IPv4 or IPv6 address literal. Validated with InetAddress.ofLiteral: parses literals only,
 * never performs a DNS lookup (unlike InetAddress.getByName, which would resolve a host name).
 * Null is considered valid: combine with @NotNull.
 */
@Documented
@Constraint(validatedBy = IpAddressValidator.class)
@Target({FIELD, METHOD, PARAMETER, ANNOTATION_TYPE, TYPE_USE})
@Retention(RUNTIME)
public @interface IpAddress {

  String message() default "must be a valid IPv4 or IPv6 address";

  Class<?>[] groups() default {};

  Class<? extends Payload>[] payload() default {};
}

package com.tipico.demo.logintracking.api.validation;

import jakarta.validation.ConstraintValidator;
import jakarta.validation.ConstraintValidatorContext;

import java.net.InetAddress;

public class IpAddressValidator implements ConstraintValidator<IpAddress, String> {

  @Override
  public boolean isValid(String value, ConstraintValidatorContext context) {
    // Null is @NotNull's business: one constraint, one rule.
    if (value == null) {
      return true;
    }
    try {
      InetAddress.ofLiteral(value);
      return true;
    } catch (IllegalArgumentException e) {
      return false;
    }
  }
}

  if [ -f "$TMP_DOWNLOAD_DIR/$distributionUrlNameMain/bin/$MVN_CMD" ]; then
    actualDistributionDir="$distributionUrlNameMain"
  fi
fi

  set +f
  for dir in "$TMP_DOWNLOAD_DIR"/*; do
    if [ -d "$dir" ]; then
      if [ -f "$dir/bin/$MVN_CMD" ]; then
        actualDistributionDir="$(basename "$dir")"
        break
      fi
    fi
  done
  set -f

  verbose "$(ls -la "$TMP_DOWNLOAD_DIR")"
  die "Could not find Maven distribution directory in extracted archive"

import com.tipico.demo.logintracking.api.TrackLoginApi;
import org.apache.hc.client5.http.impl.classic.HttpClientBuilder;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.boot.http.client.ClientHttpRequestFactoryBuilder;
import org.springframework.boot.http.client.HttpClientSettings;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.http.converter.FormHttpMessageConverter;
import org.springframework.security.oauth2.client.AuthorizedClientServiceOAuth2AuthorizedClientManager;
import org.springframework.security.oauth2.client.OAuth2AuthorizedClientManager;
import org.springframework.security.oauth2.client.OAuth2AuthorizedClientProviderBuilder;
import org.springframework.security.oauth2.client.OAuth2AuthorizedClientService;
import org.springframework.security.oauth2.client.endpoint.OAuth2AccessTokenResponseClient;
import org.springframework.security.oauth2.client.endpoint.OAuth2ClientCredentialsGrantRequest;
import org.springframework.security.oauth2.client.endpoint.RestClientClientCredentialsTokenResponseClient;
import org.springframework.security.oauth2.client.http.OAuth2ErrorResponseErrorHandler;
import org.springframework.security.oauth2.client.registration.ClientRegistrationRepository;
import org.springframework.security.oauth2.client.web.client.OAuth2ClientHttpRequestInterceptor;
import org.springframework.security.oauth2.core.http.converter.OAuth2AccessTokenResponseHttpMessageConverter;
import org.springframework.web.client.RestClient;
import org.springframework.web.client.RestClientException;
import org.springframework.web.client.support.RestClientHttpServiceGroupConfigurer;
import org.springframework.web.service.registry.ImportHttpServices;

import java.time.Duration;

/**
 * HTTP client for track-login-service, based on HTTP Service Clients (Spring Framework 7 / Spring Boot 4).
 * Spring creates the TrackLoginApi proxy from the interface and configures it directly from
 * spring.http.serviceclient.track-login.* (base URL, timeouts) — no manual RestClient,
 * HTTP client or properties classes needed. Only custom behaviour is defined here:
 * no HTTP-library retries, the success status, and the OAuth2 access token on every request.
 */
@Configuration
@ImportHttpServices(group = TrackLoginClientConfig.HTTP_SERVICE_GROUP, types = TrackLoginApi.class)
public class TrackLoginClientConfig {

  // Key of spring.http.serviceclient.<group> in application.yaml.
  static final String HTTP_SERVICE_GROUP = "track-login";

  // Key of spring.security.oauth2.client.registration.<id> in application.yaml (internal name, not known to Keycloak).
  public static final String CLIENT_REGISTRATION_ID = "track-login";

  // Retries are handled by Resilience4j only; Apache HttpClient's own retries
  // (e.g. automatic repeat on 503) would multiply attempts.
  // Used by the track-login group and by the OAuth2 token client (see tokenResponseClient).
  @Bean
  ClientHttpRequestFactoryBuilder<?> clientHttpRequestFactoryBuilder() {
    return ClientHttpRequestFactoryBuilder.httpComponents()
      .withHttpClientCustomizer(HttpClientBuilder::disableAutomaticRetries);
  }

  // Task requirement: only HTTP 200 counts as success; anything else is an error.
  @Bean
  RestClientHttpServiceGroupConfigurer trackLoginStatusHandler() {
    return groups -> groups.filterByName(HTTP_SERVICE_GROUP)
      .forEachClient((_, builder) ->
        // Treats every non-error status except 200 (e.g. 204, 3xx) as a failure (task: only 200 is success).
        // 4xx and 5xx bypass this handler and are handled by Spring's default error handling.
        builder.defaultStatusHandler(
          status -> status.value() != 200 && !status.isError(),
          (_, response) -> {
            throw new RestClientException(
              "track-login-service responded with " + response.getStatusCode().value() + ", expected 200");
          }));
  }

  /**
   * Attaches an OAuth2 access token (client credentials) to every request of the track-login group.
   * Configured here and not with @ClientRegistrationId on TrackLoginApi: the interface is the shared contract,
   * and how one client authenticates is not part of it (same reason as for the resilience annotations).
   */
  @Bean
  RestClientHttpServiceGroupConfigurer trackLoginOAuth2(OAuth2AuthorizedClientManager manager) {
    var interceptor = new OAuth2ClientHttpRequestInterceptor(manager);
    // One protected downstream per group: every request of this group uses this registration.
    interceptor.setClientRegistrationIdResolver(_ -> CLIENT_REGISTRATION_ID);
    return groups -> groups.filterByName(HTTP_SERVICE_GROUP)
      .forEachClient((_, builder) -> builder.requestInterceptor(interceptor));
  }

  /**
   * Service-based manager: calls come from Kafka listener threads, there is no incoming HTTP request.
   * The default manager stores tokens in the HTTP session and would fail here.
   * Tokens are kept in memory and reused until shortly before expiry.
   */
  @Bean
  OAuth2AuthorizedClientManager authorizedClientManager(
    ClientRegistrationRepository registrations,
    OAuth2AuthorizedClientService authorizedClients,
    ClientHttpRequestFactoryBuilder<?> requestFactoryBuilder,
    @Value("${app.oauth2.token-client.connect-timeout}") Duration connectTimeout,
    @Value("${app.oauth2.token-client.read-timeout}") Duration readTimeout) {

    var tokenResponseClient = tokenResponseClient(
      requestFactoryBuilder, HttpClientSettings.defaults().withTimeouts(connectTimeout, readTimeout));

    var manager = new AuthorizedClientServiceOAuth2AuthorizedClientManager(registrations, authorizedClients);
    manager.setAuthorizedClientProvider(OAuth2AuthorizedClientProviderBuilder.builder()
      .clientCredentials(clientCredentials -> clientCredentials.accessTokenResponseClient(tokenResponseClient))
      .build());
    return manager;
  }

  /**
   * Token requests go through Spring Security's own RestClient, not through the track-login group,
   * so the group settings do not apply to them. By default it uses Apache HttpClient with its own retries
   * (multiplying Resilience4j attempts) and without timeouts (a hanging authorization server would block
   * the listener thread). Same rules as for the API call: the same request factory builder (no library retries)
   * and explicit timeouts. Converters and error handler are the ones Spring Security requires for a custom RestClient.
   */
  private static OAuth2AccessTokenResponseClient<OAuth2ClientCredentialsGrantRequest> tokenResponseClient(
    ClientHttpRequestFactoryBuilder<?> requestFactoryBuilder, HttpClientSettings settings) {

    var restClient = RestClient.builder()
      .requestFactory(requestFactoryBuilder.build(settings))
      .configureMessageConverters(converters -> converters
        .disableDefaults()
        .addCustomConverter(new FormHttpMessageConverter())
        .addCustomConverter(new OAuth2AccessTokenResponseHttpMessageConverter()))
      .defaultStatusHandler(new OAuth2ErrorResponseErrorHandler())
      .build();

    var tokenResponseClient = new RestClientClientCredentialsTokenResponseClient();
    tokenResponseClient.setRestClient(restClient);
    return tokenResponseClient;
  }

}

/**
 * Client of track-login-service used by the application.
 * Services depend on this class, not on the TrackLoginApi proxy directly:
 * the client owns the resilience of the call (retry, later circuit breaker and bulkhead),
 * so the login-api contract stays free of Resilience4j.
 * A failed call ends with a RestClientException.
 * The OAuth2 access token is attached by the HTTP client (see TrackLoginClientConfig).
 */
@Service
public class TrackLoginClient {

  private final TrackLoginApi trackLoginApi;

  public TrackLoginClient(TrackLoginApi trackLoginApi) {
    this.trackLoginApi = trackLoginApi;
  }

  /**
   * Sends the event to track-login-service. Returns normally only on HTTP 200.
   * Transient failures (5xx, timeouts, connection errors, no access token) are retried: 4 attempts in total.
   *
   * @throws RestClientException when the call does not succeed. Subtypes:
   *         HttpServerErrorException / ResourceAccessException - after all retries failed;
   *         HttpClientErrorException (4xx) - immediately, not retried;
   *         RestClientException itself - non-200 success status (e.g. 204), not retried.
   * @throws OAuth2AuthorizationException when no access token could be obtained, after all retries failed.
   * @throws CallNotPermittedException when the circuit breaker is open (call not attempted).
   */
  @Retry(name = "track-login")
  @CircuitBreaker(name = "track-login")
  public void track(LoginEvent event) {
    trackLoginApi.trackLogin(event);
  }
}

import org.apache.kafka.common.TopicPartition;

import org.springframework.kafka.core.ProducerFactory;
import org.springframework.kafka.listener.ConcurrentMessageListenerContainer;
import org.springframework.kafka.listener.DeadLetterPublishingRecoverer;
import org.springframework.kafka.listener.DefaultErrorHandler;
import org.springframework.kafka.support.serializer.DelegatingByTypeSerializer;
import org.springframework.kafka.support.serializer.JacksonJsonSerializer;
import org.springframework.util.backoff.FixedBackOff;

import java.time.Duration;
import java.util.LinkedHashMap;
import java.util.Map;

/**
 * Configuration of the Kafka listener container:
 * <ul>
 *   <li>error handling: records that cannot be processed go to the DLT instead of blocking the partition
 *       or being skipped silently (tracking failures are not errors here, they are the unsuccessful result);</li>
 *   <li>graceful shutdown: on stop, the record in progress is finished and committed.</li>
 * </ul>
 */
@Configuration
public class KafkaListenerConfig {

  // Graceful shutdown: how long the container's stop() waits for the record in progress.
  // Not available as a Spring Boot property, hence the customizer.
  @Bean
  ContainerCustomizer<Object, Object, ConcurrentMessageListenerContainer<Object, Object>> gracefulShutdownCustomizer(
    @Value("${app.kafka.listener-shutdown-timeout}") Duration shutdownTimeout) {
    return container -> container.getContainerProperties()
      .setShutdownTimeout(shutdownTimeout.toMillis());
  }

  @Bean
  DefaultErrorHandler kafkaErrorHandler(ProducerFactory<?, ?> producerFactory,
                                        @Value("${app.kafka.topics.customer-login-dlt}") String dltTopic) {
    // Partition -1: the producer chooses by key, so the DLT does not need the same partition count as the source.
    var recoverer = new DeadLetterPublishingRecoverer(dltTemplate(producerFactory),
      (_, _) -> new TopicPartition(dltTopic, -1));
    // Option, not active: drop the stack-trace header (size; exception messages may contain fragments
    // of the payload, i.e. personal data). Type and message of the exception stay in the headers.
    // recoverer.excludeHeader(DeadLetterPublishingRecoverer.HeaderNames.HeadersToAdd.EX_STACKTRACE);
    // No retries: these errors are state-related (bad data, bug, config), a retry would fail the same way.
    return new DefaultErrorHandler(recoverer, new FixedBackOff(0, 0));
  }

  // Not a bean on purpose: a second KafkaTemplate bean would switch off Spring Boot's auto-configured one.
  private static KafkaTemplate<String, Object> dltTemplate(ProducerFactory<?, ?> producerFactory) {
    // Failed deserialization -> original bytes unchanged; failed handler -> the event as JSON.
    Map<Class<?>, Serializer<?>> serializers = new LinkedHashMap<>();
    serializers.put(byte[].class, new ByteArraySerializer());
    serializers.put(Object.class, new JacksonJsonSerializer<>());
    // Same producer settings as the application (servers, no type headers), only the serializers differ.
    return new KafkaTemplate<>(new DefaultKafkaProducerFactory<>(producerFactory.getConfigurationProperties(),
      new StringSerializer(), new DelegatingByTypeSerializer(serializers, true)));
  }
}

package com.tipico.demo.logintracking.processing.publisher;

import com.tipico.demo.logintracking.api.dto.LoginTrackingResult;
import edu.umd.cs.findbugs.annotations.SuppressFBWarnings;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.kafka.core.KafkaTemplate;
import org.springframework.stereotype.Service;

@Service
public class TrackingResultPublisher {

  @SuppressFBWarnings(value = "EI_EXPOSE_REP2",
    justification = "KafkaTemplate is a shared Spring bean injected by the container, not internal state of this class. "
      + "It must be stored as is; a defensive copy is neither needed nor possible.")
  private final KafkaTemplate<String, LoginTrackingResult> kafkaTemplate;
  private final String topic;

  public TrackingResultPublisher(KafkaTemplate<String, LoginTrackingResult> kafkaTemplate,
                                 @Value("${app.kafka.topics.login-tracking-result}") String topic) {
    this.kafkaTemplate = kafkaTemplate;
    this.topic = topic;
  }

  public void publish(LoginTrackingResult result) {
    kafkaTemplate.send(
      topic,
      // Key = customerId: Kafka routes records with the same key to the same partition,
      // so all results of one customer are delivered to consumers in publish order.
      result.customerId().toString(),
      result)
      // join(): wait for the broker ack, so a failed publish throws here instead of being lost silently.
      // The producer has already retried internally (up to delivery.timeout.ms); the listener's error handler
      // then sends the event to the DLT for replay (track-login-service is idempotent by messageId).
      // If the DLT publish fails too (Kafka down), the offset is not committed and the event is redelivered.
      .join();
  }
}

/**
 * Security of incoming HTTP requests to this service (service level, not client level).
 * <p>
 * The service exposes no API of its own; inbound traffic is only the runtime: the container healthcheck
 * (readiness probe) and the Prometheus scrape. These endpoints stay open, everything else is denied.
 * <p>
 * Why this class exists: the OAuth2 client starter (needed for outgoing calls to track-login-service)

        .requestMatchers("/actuator/health/**", "/actuator/prometheus").permitAll()
        .anyRequest().denyAll())
      // No sessions and no browser clients: nothing to protect with CSRF tokens.
      .sessionManagement(session -> session.sessionCreationPolicy(SessionCreationPolicy.STATELESS))
      .csrf(AbstractHttpConfigurer::disable)
      .build();
  }
}

package com.tipico.demo.logintracking.processing.service;

import com.tipico.demo.logintracking.api.dto.LoginEvent;
import com.tipico.demo.logintracking.api.dto.LoginTrackingResult;
import com.tipico.demo.logintracking.api.dto.LoginTrackingResult.Result;
import com.tipico.demo.logintracking.processing.client.TrackLoginClient;
import com.tipico.demo.logintracking.processing.publisher.TrackingResultPublisher;
import io.github.resilience4j.circuitbreaker.CallNotPermittedException;
import org.apache.logging.log4j.LogManager;
import org.apache.logging.log4j.Logger;
import org.springframework.security.oauth2.core.OAuth2AuthorizationException;
import org.springframework.stereotype.Service;
import org.springframework.web.client.RestClientException;

@Service
public class LoginTrackingService {

  private static final Logger log = LogManager.getLogger();

  private final TrackLoginClient trackLoginClient;
  private final TrackingResultPublisher publisher;

  public LoginTrackingService(TrackLoginClient trackLoginClient, TrackingResultPublisher publisher) {
    this.trackLoginClient = trackLoginClient;
    this.publisher = publisher;
  }

  public void process(LoginEvent event) {
    publisher.publish(LoginTrackingResult.of(event, track(event)));
  }

  // Task requirement: the result is successful only if the call succeeded (HTTP 200, retries included).
  private Result track(LoginEvent event) {
    try {
      trackLoginClient.track(event);
      return Result.successful;
    } catch (CallNotPermittedException e) {
      // TODO Production: apply backpressure instead of publishing unsuccessful while the breaker is open.
      //  Pause the Kafka listener container on transition to OPEN and resume on HALF_OPEN/CLOSED
      //  (CircuitBreaker event publisher + KafkaListenerEndpointRegistry by listener id), and re-deliver
      //  the in-flight event via the listener error handler (seek back, no skip). Events then wait in Kafka
      //  and get real attempts once the dependencies recover. The breaker records both track-login-service
      //  and authorization server failures, so one mechanism covers both. Simplified here for the assessment.
      log.warn(() -> "Tracking skipped, circuit breaker open: messageId=%s".formatted(event.messageId()));
    } catch (OAuth2AuthorizationException e) {
      // No access token (after retries): the authorization server is unreachable or rejected the token request.
      // The call to track-login-service was not made; same outcome as an unhealthy track-login-service.
      log.warn(() -> "Tracking failed, no access token: messageId=%s, reason=%s"
        .formatted(event.messageId(), e.getMessage()));
    } catch (RestClientException e) {
      // RestClientException covers every failed call to track-login-service (after retries, if any):
      // - non-200 success status (e.g. 204), thrown by our status handler;
      // - 4xx: HttpClientErrorException;
      // - 5xx: HttpServerErrorException;
      // - timeout / connection error: ResourceAccessException.
      // Any other exception is unexpected (e.g. a bug) and is not caught here.
      log.warn(() -> "Tracking failed: messageId=%s, reason=%s".formatted(event.messageId(), e.getMessage()));
    }
    return Result.unsuccessful;
  }
}

# All durations of the processing chain. Formula (property names as below in this file):
#
#   attempt  = token.connect + token.read + call.connect + call.read
#              (app.oauth2.token-client.*, spring.http.serviceclient.track-login.*)        = 1 + 2 + 2 + 3   = 8s
#   backoff  = sum over retries of wait-duration x multiplier^n x (1 + randomized-wait-factor)
#              (resilience4j.retry.instances.track-login.*)                                = 0.75 + 1.5 + 3  = 5.25s
#   record   = max-attempts x attempt + backoff                                            = 4 x 8 + 5.25    = 37.25s
#   poll     = max-poll-records x record  <  max.poll.interval.ms                          = 5 x 37.25 ~ 3.1 min < 5 min
#   shutdown : record  <  listener-shutdown-timeout  <  timeout-per-shutdown-phase  <  stop_grace_period (docker-compose.yml)
#              37.25s  <  40s                        <  45s                         <  50s
#
# Changing a value: recompute its line and every line below it.
# --- One attempt: token request ---
app:
  oauth2:
    # HTTP client for token requests to the authorization server (Keycloak), separate from the track-login client.
    # Part of the worst case per record: changing these values means rechecking the graceful shutdown chain.
    token-client:
      connect-timeout: 1s
      # A token response is small and fast; a slow authorization server should fail the attempt, not block it.
      read-timeout: 2s

  # --- Graceful shutdown: listener ---
  kafka:
    # Graceful shutdown: how long the listener waits for the record in progress on stop.
    # Must cover the worst case per record (~37.25s: 4 attempts x (token 3s + call 5s) + 5.25s backoff)
    # and stay below spring.lifecycle.timeout-per-shutdown-phase (45s).
    # Raised from 30s when the OAuth2 token request was added to each attempt.
    listener-shutdown-timeout: 40s

spring:
  # --- One attempt: call to track-login-service ---
  http:
    serviceclient:
      track-login:
        # Max time to establish a TCP connection.
        connect-timeout: 2s
        # Max time to wait for the response; keeps a slow service from blocking the listener.
        read-timeout: 3s

  kafka:
    listener:
      # Graceful shutdown: on stop, finish only the record in progress and commit it, instead of the whole poll
      # (5 x 37.25s would not fit a reasonable stop timeout). The rest is re-polled after the restart.
      immediate-stop: true

    # --- One poll ---
    consumer:
      # Worst case per record: 4 attempts x (token 1s connect + 2s read, call 2s connect + 3s read) + 5.25s backoff
      # ~ 37.25s (the token request was added with OAuth2; usually the token comes from the cache).
      # All records of one poll must finish within max.poll.interval.ms (5 min), otherwise the consumer
      # is kicked out of the group (rebalance, duplicates). 5 x 37.25s ~ 3.1 min still leaves a margin.
      max-poll-records: 5
      properties:
        # Kafka default, set explicitly because max-poll-records above is calculated against it.
        max.poll.interval.ms: 300000

#    # --- Publishing (not set yet: Kafka defaults apply, see Known limitations) ---
#    # send().join() waits up to max.block.ms (metadata / full buffer) + delivery.timeout.ms (send incl. producer retries).
#    # Defaults: 60s + 120s = 3 min per publish, far beyond the graceful shutdown chain.
#    # Values below bound one publish to ~15s; adding them means: worst case per record = 37.25s + 15s ~ 52s,
#    # so listener-shutdown-timeout, timeout-per-shutdown-phase and stop_grace_period must be raised too.
#    producer:
#      properties:
#        # Wait for metadata or free buffer space before send() returns.
#        max.block.ms: 5000
#        # Wait for one broker response.
#        request.timeout.ms: 3000
#        # Total time for one record, producer retries included; must be >= linger.ms + request.timeout.ms.
#        delivery.timeout.ms: 10000

  # --- Graceful shutdown: Spring ---
  lifecycle:
    # How long Spring waits for each shutdown phase; the listener containers stop in one of them.
    # Must exceed the listener shutdown timeout (40s); Docker's stop_grace_period must in turn exceed this.
    # Raised from 35s together with the whole chain when the token request was added to the worst case.
    timeout-per-shutdown-phase: 45s

# --- One record: attempts and backoff ---
resilience4j:
  retry:
    instances:
      # Name used in @Retry(name = "track-login") on TrackLoginClient. What is retried: see http-client.yaml.
      track-login:
        # Task requirement: 1 call + 3 retries.
        max-attempts: 4
        # Pause before each retry: around 500 ms, then 1 s, then 2 s (exponential backoff).
        wait-duration: 500ms
        enable-exponential-backoff: true
        exponential-backoff-multiplier: 2
        # Jitter: each pause is randomized by up to +-50% (worst case 0.75s, 1.5s, 3s), so threads and instances
        # do not retry in lockstep and hit a recovering service in synchronized waves (retry storm).
        enable-randomized-wait: true

    // Calls 1-2: 8 failures, below minimum-number-of-calls (10) -> the breaker stays closed.
    assertThatThrownBy(() -> trackLoginClient.track(loginEvent())).isInstanceOf(HttpServerErrorException.class);

    // so unlike the test reader in TrackingResultPublisherIT nothing is configured here.

    // Same key for both -> same partition: the valid event is queued right behind the broken one.
    send("poison-pill", "not a json");
    send("poison-pill", LOGIN_EVENT_JSON);

    // The partition is not blocked: the event after the broken one is processed.
    verify(loginTrackingService, timeout(10_000)).process(loginEvent());

    // The broken message is kept unchanged for analysis and replay.
    assertThat(readDltRecord("poison-pill").value()).isEqualTo("not a json");

    doThrow(new IllegalStateException("unexpected"))

    // The event itself is kept as JSON, ready for replay.
    JSONAssert.assertEquals(LOGIN_EVENT_JSON, dltRecord.value(), true);

    assertThat(new String(dltRecord.headers().lastHeader(KafkaHeaders.DLT_EXCEPTION_CAUSE_FQCN).value(), UTF_8))

      producer.send(new ProducerRecord<>(topic, key, value)).get();

      ConsumerConfig.KEY_DESERIALIZER_CLASS_CONFIG, StringDeserializer.class,

          .findFirst(),

  @Test
  void publishesSuccessfulResultWhenTrackingSucceeds() throws Exception {
    // Own messageId: the result topic is shared by all tests of this class.
    String messageId = UUID.randomUUID().toString();
    stubFor(post(urlEqualTo(TrackLoginApi.PATH)).willReturn(ok()));

    send(CUSTOMER_ID, loginEventJson(messageId));

    ConsumerRecord<String, String> record = awaitResult(resultJson(messageId, "successful"));
    assertThat(record.key()).isEqualTo(CUSTOMER_ID);

    // The result is published only after the REST call, so the request has already happened here.
    verify(1, postRequestedFor(urlEqualTo(TrackLoginApi.PATH))
      .withHeader("Authorization", equalTo("Bearer " + ACCESS_TOKEN))
      .withRequestBody(equalToJson(loginEventJson(messageId))));
  }

  @Test
  void publishesUnsuccessfulResultWhenTrackingFails() throws Exception {
    String messageId = UUID.randomUUID().toString();
    // 400 is a permanent error: no retries, the result is unsuccessful right away.
    stubFor(post(urlEqualTo(TrackLoginApi.PATH)).willReturn(badRequest()));

    send(CUSTOMER_ID, loginEventJson(messageId));

    ConsumerRecord<String, String> record = awaitResult(resultJson(messageId, "unsuccessful"));
    assertThat(record.key()).isEqualTo(CUSTOMER_ID);

    // Exactly one request: the permanent error was not retried.
    verify(1, postRequestedFor(urlEqualTo(TrackLoginApi.PATH)));
  }

  @Test
  void publishesUnsuccessfulResultWhenNoAccessTokenCanBeObtained() throws Exception {
    String messageId = UUID.randomUUID().toString();
    // Authorization server unavailable: handled like an unhealthy track-login-service (retried, then unsuccessful),
    // not as an unprocessable event (no DLT).
    stubFor(post(urlEqualTo(TOKEN_PATH)).willReturn(serviceUnavailable()));

    send(CUSTOMER_ID, loginEventJson(messageId));

    ConsumerRecord<String, String> record = awaitResult(resultJson(messageId, "unsuccessful"));
    assertThat(record.key()).isEqualTo(CUSTOMER_ID);

    // All attempts were spent on the token; track-login-service was never called.
    verify(4, postRequestedFor(urlEqualTo(TOKEN_PATH)));
    verify(0, postRequestedFor(urlEqualTo(TrackLoginApi.PATH)));
  }

  // Acts as the external login producer: raw JSON, no type headers.
  private void send(String key, String value) throws Exception {
    try (KafkaProducer<String, String> producer = new KafkaProducer<>(Map.of(
      ProducerConfig.BOOTSTRAP_SERVERS_CONFIG, kafka.getBootstrapServers(),
      ProducerConfig.KEY_SERIALIZER_CLASS_CONFIG, StringSerializer.class,
      ProducerConfig.VALUE_SERIALIZER_CLASS_CONFIG, StringSerializer.class))) {

      // get(): explicitly wait for the broker acknowledgement instead of relying on close() implicitly flushing the send.
      producer.send(new ProducerRecord<>(loginTopic, key, value)).get();
    }
  }

  // Acts as the external result consumer, reading raw strings.
  // The result topic is shared by all tests of this class, so each test waits for its own expected result.
  private ConsumerRecord<String, String> awaitResult(String expectedJson) {
    try (KafkaConsumer<String, String> consumer = new KafkaConsumer<>(Map.of(
      ConsumerConfig.BOOTSTRAP_SERVERS_CONFIG, kafka.getBootstrapServers(),
      // A new group per call: no committed offset, so with "earliest" it reads the topic from the beginning.
      ConsumerConfig.GROUP_ID_CONFIG, "result-reader-" + UUID.randomUUID(),
      ConsumerConfig.AUTO_OFFSET_RESET_CONFIG, "earliest",
      ConsumerConfig.KEY_DESERIALIZER_CLASS_CONFIG, StringDeserializer.class,
      ConsumerConfig.VALUE_DESERIALIZER_CLASS_CONFIG, StringDeserializer.class))) {

      consumer.subscribe(List.of(resultTopic));
      // Polls until the record arrives (the timeout is only the upper bound).
      return await().atMost(Duration.ofSeconds(10)).until(
        () -> StreamSupport.stream(consumer.poll(Duration.ofMillis(200)).spliterator(), false)
          .filter(record -> sameJson(expectedJson, record.value()))
          .findFirst(),
        Optional::isPresent).get();
    }
  }

  // Strict JSON equality (field order and formatting ignored, no extra or missing fields).
  private static boolean sameJson(String expected, String actual) {
    try {
      return JSONCompare.compareJSON(expected, actual, JSONCompareMode.STRICT).passed();
    } catch (JSONException e) {
      return false;
    }
  }
}

class LoginTrackingServiceTest {

  private final TrackLoginClient trackLoginClient = mock(TrackLoginClient.class);
  private final TrackingResultPublisher publisher = mock(TrackingResultPublisher.class);
  private final LoginTrackingService service = new LoginTrackingService(trackLoginClient, publisher);

  private final LoginEvent event = loginEvent();

  @Test
  void publishesSuccessfulWhenTrackingSucceeds() {
    // Given: mocked void method does nothing by default, i.e. the REST call "succeeds" (HTTP 200)
    service.process(event);
    // Then: the result event carries all input fields plus successful
    verify(publisher).publish(LoginTrackingResult.of(event, Result.successful));
  }

  @Test
  void publishesUnsuccessfulWhenTrackingFails() {
    // Given: the client gives up (retries exhausted or non-retryable error) and throws RestClientException
    doThrow(new RestClientException("track-login-service unavailable")).when(trackLoginClient).track(event);

    service.process(event);

    // Then: the failure is translated into unsuccessful instead of propagating to the listener
    verify(publisher).publish(LoginTrackingResult.of(event, Result.unsuccessful));
  }

  /** Unexpected errors (bugs) are not turned into unsuccessful: they propagate, so the Kafka offset is not committed. */
  @Test
  void propagatesUnexpectedErrorsWithoutPublishing() {
    // Given: an error that is not a failed HTTP call (e.g. a bug)
    doThrow(new IllegalStateException("bug")).when(trackLoginClient).track(event);

    // Then: it is not swallowed as unsuccessful ...
    assertThatThrownBy(() -> service.process(event)).isInstanceOf(IllegalStateException.class);

    // ... and no result is published
    verify(publisher, never()).publish(any());
  }

  @Test
  void publishesUnsuccessfulWhenCircuitBreakerIsOpen() {
    // Given: the breaker is open, so the client fails fast without calling track-login-service
    doThrow(CallNotPermittedException.createCallNotPermittedException(CircuitBreaker.ofDefaults("track-login")))
      .when(trackLoginClient).track(event);

    service.process(event);

    // Then: simplified behaviour for the assessment (see TODO in the service): treated as unsuccessful
    verify(publisher).publish(LoginTrackingResult.of(event, Result.unsuccessful));
  }

  str="${1:-}" h=0

    str="${str#?}"
  done
  printf %x\\n $h

# Login Tracking Parent

Shared Maven parent POM for all child projects.

It plays the role of a corporate parent POM: build rules and code quality checks are defined once here and apply 
**automatically** to every child project, **without any configuration** in the children.

## Inheritance chain

```
spring-boot-starter-parent        dependency & plugin versions, compiler defaults
        ↑
login-tracking-parent             Java version, build rules, quality gates
        ↑
   child projects
```

## What children get

- **Java 25** as the compilation target (`java.version`).
- Everything from `spring-boot-starter-parent`: 
  - managed dependency versions,
  - plugin versions, 
  - UTF-8 encoding, 
  - `-parameters` compiler flag.
- Mandatory quality gates listed below.

## Plugins

| Plugin | Purpose |
|---|---|
| **maven-enforcer-plugin** | Fails the build immediately with a clear message if the JDK is older than `java.version`. |
| **jacoco-maven-plugin** | Measures test coverage, generates a report and fails the build if line coverage is below **90%** (`jacoco.minimum.coverage`). |
| **spotbugs-maven-plugin** | Static analysis on bytecode to find real defects: possible NPEs, broken `equals`/`hashCode`, resource and concurrency issues. |
| **findsecbugs-plugin** | SpotBugs extension for security issues: injections, weak cryptography, unsafe deserialization, etc. |
| **maven-pmd-plugin** | Code quality rules: best practices, design issues, error-prone patterns, dead code. |

## Running and skipping checks

All quality gates are bound to the `verify` phase:

| Command | Quality gates |
|---|---|
| `./mvnw test` | not executed |
| `./mvnw clean package` | not executed |
| `./mvnw verify` | executed |
| `./mvnw clean install` | executed |

To skip the checks on `install` (local iteration only, the final build must pass them):

```bash
./mvnw clean install -Dspotbugs.skip -Dpmd.skip -Djacoco.skip
```

## Reports

Generated in each child project after `./mvnw verify` (or `install`):

| Tool | Location |
|---|---|
| JaCoCo | `target/site/jacoco/index.html` (HTML), `target/site/jacoco/jacoco.xml` |
| SpotBugs | `target/spotbugsXml.xml`; interactive view: `./mvnw spotbugs:gui` |

## Overview

```

          │
          ▼
 login-processing-service ── client credentials ──▶ Keycloak (access token, cached)
          │
          │  POST /api/v1/logins, Bearer token, up to 4 attempts
```

| PostgreSQL                 | Storage of track-login-service                                                                |

| # | Requirement                                                                                         | Implementation                                                           | Proven by                                                                        |
|---|-----------------------------------------------------------------------------------------------------|--------------------------------------------------------------------------|----------------------------------------------------------------------------------|
| 1 | Event attributes: customerId, username, client (web/android/ios), timestamp, messageId, customerIp | `LoginEvent` in `login-api`, validated in track-login-service            | `LoginEventSerializationTest`, `TrackLoginControllerTest`                        |
| 2 | Result is `successful` or `unsuccessful`                                                            | `LoginTrackingResult.Result`                                             | `LoginTrackingResultSerializationTest`                                           |
| 3 | Successful only on HTTP 200                                                                         | Status handler in `TrackLoginClientConfig` (e.g. 204 is a failure)       | `TrackLoginApiTest`                                                              |
| 4 | Retry 3 times                                                                                       | Resilience4j retry, 4 attempts in total, exponential backoff with jitter | `TrackLoginClientTest`                                                           |
| 5 | Unsuccessful when the 4th attempt fails                                                             | `LoginTrackingService`                                                   | `TrackLoginClientTest`, `LoginTrackingServiceTest`, `LoginProcessingEndToEndIT`  |

retried (5xx, timeouts, connection errors, no access token). A 4xx means the request itself is wrong; repeating it
cannot succeed, so the result is `unsuccessful` right away. The other resilience mechanisms follow the same idea:
see [How it works](#how-it-works).

## Architecture and technologies

### Project structure

```
login-tracking/
├── build.sh                    builds all projects in dependency order
├── docker-compose.yml          the whole system locally (+ docker-compose.debug.yml, .env)
├── infrastructure/             Keycloak realm, Prometheus, Grafana, Alloy, healthcheck script, WireMock stubs
├── login-tracking-parent/      parent POM
├── login-api/                  shared contract
├── login-processing-service/
└── track-login-service/
```

**Independent builds, not a multi-module project.** Each project has its own Maven build and version, as if it were
a separate repository: a service is built, tested and released on its own. `build.sh` only runs the builds in
dependency order (parent → contract → services).

**Shared parent POM.** Like a corporate parent: Java version, quality gates (JaCoCo 90%, SpotBugs + FindSecBugs, PMD,
JDK check) and the integration test profile are defined once and apply to every project without configuration
in the children. Details: [login-tracking-parent](login-tracking-parent/README.md).

**Shared contract `login-api`.** The HTTP interface (`TrackLoginApi`), the DTOs and their validation rules.
The server implements the interface, the client calls it through a generated proxy, so path, method, JSON and
validation cannot drift apart. The contract contains no client concerns (no retry, no authentication):
those are decided by each client.

### Configuration structure

login-processing-service splits its configuration by concern instead of one large `application.yaml`:

## How it works

### Message formats

Incoming event, topic `customer-login`, key = `customerId`:
```json
{
  "customerId": "3f2b8c1e-6d4a-4f7e-9b2a-1c5d8e7f6a90",
  "username": "john.doe",
  "client": "web",
  "timestamp": "2026-11-21T15:36:09Z",
  "messageId": "b765096c-edf1-464a-b564-9f66074a126b",
  "customerIp": "192.168.1.10"
}
```

Result, topic `login-tracking-result`, key = `customerId`: all event fields plus `result` (`successful` / `unsuccessful`):
```json
{
  "customerId": "3f2b8c1e-6d4a-4f7e-9b2a-1c5d8e7f6a90",
  "username": "john.doe",
  "client": "web",
  "timestamp": "2026-11-21T15:36:09Z",
  "messageId": "b765096c-edf1-464a-b564-9f66074a126b",
  "customerIp": "192.168.1.10",
  "result": "successful"
}
```

Dead letter topic `customer-login-dlt`: the original message unchanged (raw bytes if it was not readable),
the reason in the record headers.

No Java type information in Kafka headers: producers and consumers depend on the JSON only.

### 1. Kafka listener (login-processing-service)

- One consumer group for all instances; partitions are split between them. Threads per instance from the environment
  (3 = one per partition here).
- The JSON deserializer is wrapped in `ErrorHandlingDeserializer`: a broken message reaches the error handler
  instead of failing inside `poll()` forever.
- Up to 5 records per poll; offsets are committed after processing (at-least-once delivery).

| What goes wrong                                   | What happens                                                | Handled by                                                                           |
|---------------------------------------------------|-------------------------------------------------------------|--------------------------------------------------------------------------------------|
| Message is not JSON, or an unknown `client` value | Original message goes to the DLT, the next one is processed | `ErrorHandlingDeserializer`, `DefaultErrorHandler` + `DeadLetterPublishingRecoverer` |
| Unexpected error while processing (a bug)         | Event goes to the DLT, no retries (would fail the same way) | Same error handler                                                                   |
| Service stops or crashes during a record          | Offset not committed, the event is processed again          | At-least-once + idempotent storage (step 4)                                          |

### 2. Access token (Keycloak)

- OAuth2 client credentials: the service authenticates as itself, no user involved.
- The token is requested only when missing or about to expire (60 s before), cached in memory and shared by all
  listener threads: one token request per ~4 minutes, not per event.
- Token requests use their own HTTP client: no library retries, timeouts 1 s connect / 2 s read.

| What goes wrong                   | What happens                                                                              | Handled by                       |
|-----------------------------------|-------------------------------------------------------------------------------------------|----------------------------------|
| Keycloak unreachable or too slow  | Attempt fails, retried like a failed call, then `unsuccessful`                            | Retry + circuit breaker (step 3) |
| Wrong client secret               | Every token request fails, the circuit breaker opens quickly: misconfiguration is visible | Same                             |
| Keycloak down, token still cached | Nothing: calls continue until the token expires                                           | Token cache                      |

### 3. REST call to track-login-service

- Client generated from the shared `TrackLoginApi`, Apache HttpClient 5, timeouts 2 s connect / 3 s read.
- Only HTTP 200 is success; any other status (also 204 or 3xx) is a failure.
- Retry (Resilience4j): 4 attempts in total, backoff ~0.5 s, 1 s, 2 s with ±50% jitter, transient failures only.
- Circuit breaker: over the last 10 calls, ≥ 50% failures → open for 30 s (fail fast, no calls), then 3 trial calls.

| What goes wrong                    | What happens                                                                   | Handled by                           |
|------------------------------------|--------------------------------------------------------------------------------|--------------------------------------|
| 5xx, timeout, connection refused   | Retried; after the 4th failed attempt `unsuccessful`                           | Retry                                |
| 4xx (invalid event, 401, 403)      | Not retried, `unsuccessful` right away                                         | Retry configuration (transient only) |
| Service down for longer            | Breaker opens: `unsuccessful` immediately, no requests; trial calls after 30 s | Circuit breaker                      |
| HTTP library retries on its own    | Disabled: it would multiply the attempts                                       | Apache HttpClient configuration      |

### 4. Storage (track-login-service)

- The request passes the security filter first: valid JWT from Keycloak, audience `track-login-service`,
  scope `logins:write`.
- The event is validated with the rules of the shared contract (including a custom IP address check without DNS
  lookup). Errors follow RFC 9457 (Problem Details) with a list of all invalid fields; rejected values are not echoed
  (personal data).
- The vital information is inserted; `messageId` is the primary key.
- Database budget per request: 1 s for a connection + 1.5 s per statement, below the caller's 3 s read timeout.

What is stored (data minimization):

| Field        | Stored | Why                                                                         |
|--------------|--------|-----------------------------------------------------------------------------|
| `messageId`  | yes    | Primary key: one event is stored once (idempotency by design)               |
| `customerId` | yes    | Identifies the customer; indexed for "logins of a customer"                 |
| `client`     | yes    | Restricted by a database check constraint                                   |
| `timestamp`  | yes    | When the login happened                                                     |
| `customerIp` | yes    | The most valuable field for login tracking (security, fraud)                |
| `username`   | **no** | The customer is identified by `customerId`; the name is extra personal data |
| `created_at` | yes    | Technical: when the event was stored                                        |

`username` still arrives in the request (one shared contract); the IP address is personal data too and would need
a retention period in production.

| What goes wrong                                            | What happens                                                  | Handled by                                                 |
|------------------------------------------------------------|---------------------------------------------------------------|------------------------------------------------------------|
| No or invalid token / wrong scope                          | 401 / 403                                                     | OAuth2 resource server, `SecurityConfig`                   |
| Invalid event                                              | 400 with all invalid fields                                   | Bean Validation, `RestExceptionHandler`                    |
| Same `messageId` again (redelivery, retry after a timeout) | 200, not stored twice (an existing row is never overwritten)  | Primary key + duplicate handling in `LoginTrackingService` |
| Database slow or down                                      | Fails within 2.5 s instead of blocking; the caller retries    | Hikari connection timeout, PostgreSQL `statement_timeout`  |

### 5. Publish the result (login-processing-service)

- Key = `customerId`: all results of one customer land in one partition, in order.
- The listener waits for the broker's acknowledgement before the offset is committed.

| What goes wrong                           | What happens                                                                                                          | Handled by                           |
|-------------------------------------------|-----------------------------------------------------------------------------------------------------------------------|--------------------------------------|
| Kafka briefly unavailable                 | The producer retries internally (up to 2 min)                                                                         | Kafka producer                       |
| Publish fails finally                     | Event goes to the DLT for replay; if Kafka is completely down, the offset is not committed and the event comes again | Error handler (step 1)               |
| Crash after publishing, before the commit | Processed again: a second result for the same `messageId` (at-least-once)                                             | Consumers deduplicate by `messageId` |

the results and the DLT in Kafka UI (http://localhost:8085), the service log with

| Scenario                    | Break it                                  | Expected                                                                                                                         | Restore                                                                          |
|-----------------------------|-------------------------------------------|----------------------------------------------------------------------------------------------------------------------------------|----------------------------------------------------------------------------------|
| Message is not JSON         | Send the "not JSON" message               | Message in `customer-login-dlt` with the error in its headers; the next event is processed                                      | —                                                                                |
| Invalid event (e.g. bad IP) | Send the event below                      | track-login-service answers 400, result `unsuccessful`, no retries                                                              | —                                                                                |
| track-login-service down    | `docker compose stop track-login-service` | `Tracking failed` after 4 attempts, result `unsuccessful`; after a few events `circuit breaker open`: `unsuccessful` without calls | `docker compose start track-login-service` (breaker closes after its 30 s trial) |
| Keycloak down               | `docker compose stop keycloak`            | Results stay `successful` while the cached token is valid (up to ~4 min), then `no access token`, `unsuccessful`                 | `docker compose start keycloak`                                                  |
| PostgreSQL down             | `docker compose stop postgres`            | track-login-service health `DOWN`, the call fails fast, result `unsuccessful`                                                   | `docker compose start postgres`                                                  |
| Kafka down                  | `docker compose stop kafka`               | Nothing is consumed or published; connection errors in the log                                                                  | `docker compose start kafka`; events sent afterwards are processed               |


## Timeouts and graceful shutdown

Every wait in the processing chain is bounded, and each outer limit is larger than what runs inside it.
All values of login-processing-service are in one file, with the formula on top:
`login-processing-service/src/main/resources/settings/timeouts.yaml`. A changed value means rechecking the chain.

### Worst case per event

| Term          | Formula / setting                                                                                                                         | Value                       |
|---------------|-------------------------------------------------------------------------------------------------------------------------------------------|-----------------------------|
| Token request | `app.oauth2.token-client.connect-timeout` + `read-timeout` (only if the token is not cached)                                              | 1 + 2 = 3 s                 |
| REST call     | `spring.http.serviceclient.track-login.connect-timeout` + `read-timeout`                                                                  | 2 + 3 = 5 s                 |
| One attempt   | token request + REST call                                                                                                                 | 8 s                         |
| Backoff       | `resilience4j.retry.instances.track-login`: `wait-duration` × `exponential-backoff-multiplier`ⁿ × (1 + `randomized-wait-factor`), 3 pauses | 0.75 + 1.5 + 3 = 5.25 s     |
| One event     | `max-attempts` × attempt + backoff                                                                                                        | 4 × 8 + 5.25 = 37.25 s      |
| One poll      | `spring.kafka.consumer.max-poll-records` × event, must be < `spring.kafka.consumer.properties.max.poll.interval.ms`                       | 5 × 37.25 ≈ 3.1 min < 5 min |

Inside track-login-service: `spring.datasource.hikari.connection-timeout` (1 s) + PostgreSQL `statement_timeout`
(1.5 s, in `spring.datasource.hikari.data-source-properties`) must stay below the caller's read timeout (3 s).

### Graceful shutdown

On stop the service finishes the event in progress, publishes its result, commits the offset and only then exits;
the rest of the poll is processed after the restart (`immediate-stop`). The layers wait for each other:

| Layer                 | Setting                                       | Value | Must be greater than                                  |
|-----------------------|-----------------------------------------------|-------|-------------------------------------------------------|
| Listener container    | `app.kafka.listener-shutdown-timeout`         | 40 s  | worst case per event (37.25 s)                        |
| Spring shutdown phase | `spring.lifecycle.timeout-per-shutdown-phase` | 45 s  | listener                                              |
| Docker                | `stop_grace_period` (docker-compose.yml)      | 50 s  | Spring (Docker's default 10 s would kill the process) |

Try it: freeze track-login-service so every call runs into the read timeout, send an event, stop the service
while it is retrying:
```bash
docker compose pause track-login-service
# send a valid event (Quick start), then immediately:
time docker compose stop login-processing-service
docker inspect login-processing-service --format '{{.State.ExitCode}}'
docker compose unpause track-login-service
docker compose start login-processing-service
```
Expected: the stop waits for the retries to finish; the log shows `Tracking failed`, then `partitions revoked`;
exit code 143 (SIGTERM, clean), not 137 (SIGKILL). After the start the event is not processed again.

## Authentication

Service-to-service OAuth2 with Keycloak as the authorization server.

| Role                 | Who                      | What it does                                                              |
|----------------------|--------------------------|---------------------------------------------------------------------------|
| Authorization server | Keycloak                 | Issues signed access tokens (JWT)                                         |
| Client               | login-processing-service | Gets a token with client credentials, sends it as `Authorization: Bearer` |
| Resource server      | track-login-service      | Validates the token locally, checks audience and scope                    |

**Keycloak** starts already configured: the realm is imported from `infrastructure/keycloak/realm-login-tracking.json`.
It defines the client `login-processing-service` (client credentials only), the scope `logins:write` and an audience
mapper that puts `track-login-service` into the token. The client secret comes from `.env`, shared by Keycloak and
the service.

**login-processing-service (client)**
- Gets a token only when there is none or it is about to expire; cached in memory for all listener threads.
- The token is attached by an interceptor of the HTTP client, not by the shared contract: how a client authenticates
  is not part of the API.
- **Token requests: Spring's default HTTP client replaced.** Spring Security fetches tokens with its own internal
  client, which ignores the settings of our HTTP client: Apache's default retries (one extra request on 503,
  doubling the attempts) and no read timeout, so a hanging Keycloak would block a listener thread forever.
  `TrackLoginClientConfig#tokenResponseClient` gives the token client its own `RestClient`: the same request factory
  as the REST call (library retries off) and explicit timeouts (`app.oauth2.token-client.*`).
  Proven by `TrackLoginClientTest#retriesWhenAccessTokenCannotBeObtained` (exactly 4 token requests, not 8) and
  `TrackLoginClientTokenTimeoutTest` (a slow Keycloak fails the attempt instead of blocking it).
- No token (Keycloak down, wrong secret) is handled like a failed call: retried, then `unsuccessful`.

**track-login-service (resource server)**
- Validates each token locally: signature (keys fetched once from Keycloak and cached), expiry, issuer, audience.
  No call to Keycloak per request.
- `POST /api/v1/logins` requires the scope `logins:write`: no token or invalid token → 401, valid token without the
  scope → 403. Health and Prometheus endpoints are open (the runtime has no token), everything else is denied.
- Stateless: no session, no CSRF (bearer tokens are not sent automatically by a browser).

The issuer in the token must match exactly what track-login-service expects (`http://keycloak:8080/realms/login-tracking`
inside compose). A token requested from the host via `localhost:8180` has a different issuer and is rejected.

Realm attributes and curl examples (get a token, decode it, introspection): [infrastructure/keycloak](infrastructure/keycloak/README.md).

Grafana Explore: http://localhost:3000/explore (data source Prometheus or Loki).

#### HTTP requests

HTTP requests. `rate(...[1m])` = requests per second (load, for production); `increase(...[1h])` = how many requests

Server side — what track-login-service answered, by status. Requests that never reached a stopped server are not here:

Same as requests per second:
```promql

```

Client side — what login-processing-service got when calling track-login-service, every attempt counted.
`status="CLIENT_ERROR"` = no HTTP response at all (connection refused, timeout), the reason is in `exception`.

Circuit breaker state. `job` = which service (set by Prometheus from `prometheus.yml`), `name` = the breaker's name
from the annotation (unique only within one service).

All states: one line per possible state, `1` = the state the breaker was in at that moment, `0` = all others —


Current state only (`== 1` keeps only the series that are 1; a state that was 1 at any moment in the selected range


Kafka. Consumer and producer metrics come from the Kafka client (Micrometer, registered by Spring Boot);

## Trade-offs

Decisions where more than one option was reasonable, in the order of the processing flow.

### 1. Kafka listener
- **At-least-once, offsets committed per poll** (instead of per record or exactly-once). A restart can repeat up to
  5 events; this is safe because storage is idempotent by `messageId`, and exactly-once does not cover an external
  REST call anyway.
- **Concurrency from the environment, no bulkhead.** Each event makes exactly one call, the listener is the only
  caller: the number of consumer threads already caps the load on track-login-service. A bulkhead pays off with
  several callers or several downstreams.
- **No retries before the DLT.** What lands there (unreadable message, bug) fails the same way on every retry;
  transient failures are retried earlier, at the REST call.

### 2. Access token
- **Local token validation in track-login-service** (instead of asking Keycloak per request). No latency and no runtime
  dependency on Keycloak per request; a revoked client keeps working until its token expires (5 minutes).
- **Token attached by the client's HTTP configuration**, not by an annotation on the shared interface: the contract
  describes the API, how one client authenticates is not part of it.
- **One circuit breaker for the whole call, token included.** A failing Keycloak and a failing track-login-service
  have the same effect (the call cannot be made), so one mechanism reacts to both.

### 3. REST call
- **Retry outside the circuit breaker.** Every attempt is recorded, so the breaker reacts to the real request rate
  on the downstream; the cost: the window fills four times faster, and one failing event can decide a half-open trial.
- **Open breaker = `unsuccessful`.** Simplification for the assessment. Production: pause the listener while the
  breaker is open so events wait in Kafka (needs `automatic-transition-from-open-to-half-open-enabled`, otherwise a
  paused listener never triggers the transition back).
- **No answer within the read timeout = `unsuccessful`**, even if track-login-service stored the event: the task
  defines success as receiving HTTP 200.

### 4. Storage
- **Idempotency by primary key and catching the duplicate** (instead of native `INSERT ... ON CONFLICT DO NOTHING`).
  Portable, no SQL in the repository; the cost is an error line in the Hibernate log per duplicate.
- **`Persistable` for the assigned id** (instead of `@Version` or a surrogate key): a duplicate fails instead of being
  merged, and the schema is not changed to fit the framework.
- **Validation rules in the shared contract** (instead of a server-only request DTO): one set of rules for client
  and server. The username is still sent because the contract is shared, but it is not stored.

### 5. Publishing the result
- **Synchronous publish per event** (wait for the broker's acknowledgement): a failed publish is never lost silently.
  Reliability over throughput; scaling goes through partitions and instances.
- **Results are at-least-once too:** after a crash between publishing and committing, a second result for the same
  `messageId` can follow. Consumers deduplicate by `messageId`.

### Cross-cutting
- **Independent builds instead of a multi-module project:** each service builds and releases on its own;
  shared rules come from the parent POM.
- **Configuration split by concern**, with all durations in one file: a change stays local, the timeout chain is
  checked in one place.
- **Logs collected outside the application** (Alloy reads container output) instead of a log appender in the code:
  no dependency, and a logging outage does not affect the service.

## Known limitations

This is an assessment, not a production system. The must-have cases are implemented and covered by tests.
The points below are known and left out on purpose: each needs another day or two to do properly.
Happy to discuss any of them.

| Area                 | Limitation                                                                                                                                                                                                         | How it would be solved                                                                                              |
|----------------------|--------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------|---------------------------------------------------------------------------------------------------------------------|
| Incoming events      | Events are not validated in login-processing-service. Without `customerId` the event reaches the DLT only because publishing fails on the missing record key; without `messageId` a result with empty fields is published. | Validate in the listener with the rules already in the shared contract; invalid events go to the DLT explicitly. |
| Input topic          | Per-customer order holds only if the upstream producer keys events by `customerId`.                                                                                                                               | State it in the topic contract.                                                                                     |
| Signing keys         | After a Keycloak restart (`start-dev` keeps keys in memory) or a key rotation, the cached token is rejected with 401 until it expires (up to ~4 min); 401 is not retried.                                         | Drop the cached token on 401 (failure handler of the OAuth2 interceptor) and retry once with a new token.          |
| Key download         | track-login-service fetches Keycloak's keys with Spring's default HTTP client, without timeouts (first request, key rotation).                                                                                    | Own `JwtDecoder` with a timed HTTP client, as done for the token requests.                                         |
| Kafka producer       | Producer timeouts (`delivery.timeout.ms`, `max.block.ms`, minutes by default) are not part of the timeout chain: with a hanging broker the graceful shutdown limits can be exceeded. No data is lost (the offset is not committed). | Set them explicitly and add them to `timeouts.yaml`.                                                   |
| HTTP connection pool | Default pool of Apache HttpClient: 5 connections per host. Enough for 3 listener threads; with more threads they would wait for a connection outside the timeout chain.                                           | Size the pool to the concurrency, add the pool wait to the chain.                                                  |
| Rate limiting        | 429 is not retried (track-login-service has no rate limiting).                                                                                                                                                     | Retry 429 honouring `Retry-After`.                                                                                  |
| Database             | The schema serves writes only.                                                                                                                                                                                     | Indexes, retention and time partitioning follow the read and retention requirements.                                |
| Personal data        | The DLT holds full events (username, IP); the IP is stored without a retention period.                                                                                                                            | Restricted access and retention for the DLT and the table.                                                          |
| Actuator             | Health details are public, metrics share the application port.                                                                                                                                                    | Separate management port or authentication; details only when authorized.                                          |
| Local setup          | Keycloak in dev mode, demo secrets in `.env`.                                                                                                                                                                      | Production mode with TLS and a database, secrets from a secret store.                                               |

  if [ -f "$TMP_DOWNLOAD_DIR/$distributionUrlNameMain/bin/$MVN_CMD" ]; then
    actualDistributionDir="$distributionUrlNameMain"
  fi
fi

  set +f
  for dir in "$TMP_DOWNLOAD_DIR"/*; do
    if [ -d "$dir" ]; then
      if [ -f "$dir/bin/$MVN_CMD" ]; then
        actualDistributionDir="$(basename "$dir")"
        break
      fi
    fi
  done
  set -f

  verbose "Contents of $TMP_DOWNLOAD_DIR:"
  verbose "$(ls -la "$TMP_DOWNLOAD_DIR")"
  die "Could not find Maven distribution directory in extracted archive"
fi

clean || :

import com.tipico.demo.logintracking.api.dto.LoginEvent;
import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.EnumType;
import jakarta.persistence.Enumerated;
import jakarta.persistence.Id;
import jakarta.persistence.PostLoad;
import jakarta.persistence.PostPersist;
import jakarta.persistence.Table;
import jakarta.persistence.Transient;
import jakarta.validation.constraints.NotNull;
import jakarta.validation.constraints.Size;
import org.springframework.data.domain.Persistable;

import java.time.Instant;
import java.util.UUID;

/**
 * Stored vital information of a login event (no username, see README).
 *
 * <p>The id (messageId) is assigned by the producer, not generated. Spring Data's save() would therefore treat
 * every instance as existing and merge it (SELECT + UPDATE: a duplicate would overwrite the stored row).
 * Implementing {@link Persistable} lets the entity decide itself: created in code = new = plain INSERT,
 * so a duplicate fails on the primary key instead of being merged.
 */
@Entity
@Table(name = "login_event")
public class LoginEventEntity implements Persistable<UUID> {

  @Id
  @NotNull
  private UUID messageId;

  @NotNull
  private UUID customerId;

  @NotNull
  @Enumerated(EnumType.STRING)
  private LoginEvent.Client client;

  @NotNull
  private Instant loginTimestamp;

  // 45 characters: the longest textual IPv6 form, same limit as the column.
  @NotNull
  @Size(max = 45)
  private String customerIp;

  // Set by the database (default now()), never written by the application.
  @Column(insertable = false, updatable = false)
  private Instant createdAt;

  // Not a column: true for an entity created in code, false once it is loaded from or stored in the database.
  @Transient
  private boolean isNew = true;

  // Required by JPA; application code uses from(...).
  protected LoginEventEntity() {
  }

  public static LoginEventEntity from(LoginEvent event) {
    var entity = new LoginEventEntity();
    entity.messageId = event.messageId();
    entity.customerId = event.customerId();
    entity.client = event.client();
    entity.loginTimestamp = event.timestamp();
    entity.customerIp = event.customerIp();
    return entity;
  }

  @Override
  public UUID getId() {
    return messageId;
  }

  public UUID getCustomerId() {
    return customerId;
  }

  public LoginEvent.Client getClient() {
    return client;
  }

  public Instant getLoginTimestamp() {
    return loginTimestamp;
  }

  public String getCustomerIp() {
    return customerIp;
  }

  public Instant getCreatedAt() {
    return createdAt;
  }

  @Override
  public boolean isNew() {
    return isNew;
  }

  // JPA lifecycle callbacks: after loading or after the INSERT the entity is no longer new.
  @PostLoad
  @PostPersist
  void markNotNew() {
    isNew = false;
  }
}

import org.springframework.http.HttpHeaders;
import org.springframework.http.HttpStatusCode;
import org.springframework.http.ProblemDetail;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.RestControllerAdvice;
import org.springframework.web.context.request.WebRequest;
import org.springframework.web.servlet.mvc.method.annotation.ResponseEntityExceptionHandler;

import java.util.List;
import java.util.Objects;

/**
 * Extends the standard Problem Details (RFC 9457) responses with the list of invalid fields.
 * Bean Validation collects all violations (fail slow), so the client sees every problem at once.
 * Rejected values are deliberately not echoed: they may be personal data (IP, username).
 */
@RestControllerAdvice
public class RestExceptionHandler extends ResponseEntityExceptionHandler {

  @Override
  protected ResponseEntity<Object> handleMethodArgumentNotValid(
    MethodArgumentNotValidException ex,
    HttpHeaders headers,
    HttpStatusCode status,
    WebRequest request) {
    List<FieldViolation> violations = ex.getBindingResult().getFieldErrors().stream()
      .map(error -> new FieldViolation(
        error.getField(),
        Objects.requireNonNullElse(error.getDefaultMessage(), "invalid")))
      .toList();

    ProblemDetail problem = ex.getBody();
    problem.setDetail("Validation failed for %d field(s).".formatted(violations.size()));
    problem.setProperty("errors", violations);
    return handleExceptionInternal(ex, problem, headers, status, request);
  }

  /** One invalid field: which one and which rule it breaks. */
  record FieldViolation(String field, String message) {
  }
}

package com.tipico.demo.logintracking.track.rest.security;

import com.tipico.demo.logintracking.api.TrackLoginApi;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.http.HttpMethod;
import org.springframework.security.config.Customizer;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.annotation.web.configurers.AbstractHttpConfigurer;
import org.springframework.security.config.http.SessionCreationPolicy;
import org.springframework.security.web.SecurityFilterChain;

/**
 * Security as a cross-cutting concern: filters in front of the controllers, no security code in REST or service.
 * Token validation itself is configured, not coded (spring.security.oauth2.resourceserver.jwt.*).
 */
@Configuration
public class SecurityConfig {

  // Scope granted to login-processing-service by the authorization server; Spring maps scopes to "SCOPE_..." authorities.
  private static final String WRITE_LOGINS = "SCOPE_logins:write";

  @Bean
  SecurityFilterChain securityFilterChain(HttpSecurity http) throws Exception {
    return http
      .authorizeHttpRequests(requests -> requests
        // Runtime needs: container healthcheck and Prometheus scrape. No token: the runtime has none.
        .requestMatchers("/actuator/health/**", "/actuator/prometheus").permitAll()
        .requestMatchers(HttpMethod.POST, TrackLoginApi.PATH).hasAuthority(WRITE_LOGINS)
        // Anything else: deny by default.
        .anyRequest().denyAll())
      .oauth2ResourceServer(oauth2 -> oauth2.jwt(Customizer.withDefaults()))
      // Stateless API: every request carries its token, no HTTP session.
      .sessionManagement(session -> session.sessionCreationPolicy(SessionCreationPolicy.STATELESS))
      // CSRF protects cookie-based sessions in browsers; with bearer tokens and no session it does not apply.
      .csrf(AbstractHttpConfigurer::disable)
      .build();
  }
}

package com.tipico.demo.logintracking.track.service;

import com.tipico.demo.logintracking.api.dto.LoginEvent;
import com.tipico.demo.logintracking.track.persistence.LoginEventEntity;
import com.tipico.demo.logintracking.track.persistence.LoginEventRepository;
import org.apache.logging.log4j.LogManager;
import org.apache.logging.log4j.Logger;
import org.hibernate.exception.ConstraintViolationException;
import org.springframework.dao.DataIntegrityViolationException;
import org.springframework.stereotype.Service;

/**
 * Stores login events idempotently: a duplicate messageId (redelivery, retry of the caller) is a normal outcome.
 *
 * <p>Deliberately NOT @Transactional: save() runs in its own transaction, so a duplicate fails within that call
 * and can be handled here. Inside an outer transaction the failure would surface only at its commit
 * (uncaught, 500 instead of 200), and the transaction would already be rollback-only.
 */
@Service
public class LoginTrackingService {

  private static final Logger log = LogManager.getLogger();

  // Name given to the primary key in the Liquibase changelog.
  private static final String PRIMARY_KEY = "pk_login_event";

  private final LoginEventRepository repository;

  public LoginTrackingService(LoginEventRepository repository) {
    this.repository = repository;
  }

  public void track(LoginEvent event) {
    try {
      repository.save(LoginEventEntity.from(event));
    } catch (DataIntegrityViolationException e) {
      if (!isDuplicate(e)) {
        throw e;
      }
      // Expected under at-least-once delivery: the event is already stored, nothing to do.
      log.warn(() -> "Duplicate login event ignored: messageId=%s".formatted(event.messageId()));
    }
  }

  // Only a violation of the primary key means "already stored"; any other constraint is a real error.
  private static boolean isDuplicate(DataIntegrityViolationException e) {
    for (Throwable cause = e; cause != null; cause = cause.getCause()) {
      if (cause instanceof ConstraintViolationException violation) {
        return PRIMARY_KEY.equals(violation.getConstraintName());
      }
    }
    return false;
  }
}

    // Not a duplicate but a real error: must not be swallowed.
    assertThatThrownBy(() -> service.track(loginEvent())).isSameAs(otherViolation);
  }

  @Test
  void propagatesViolationWithoutConstraintName() {
    DataIntegrityViolationException unknownViolation = new DataIntegrityViolationException("unknown");
    when(repository.save(any())).thenThrow(unknownViolation);

    assertThatThrownBy(() -> service.track(loginEvent())).isSameAs(unknownViolation);
  }

  // As Spring delivers it: a DataIntegrityViolationException wrapping Hibernate's violation with the constraint name.
  private static DataIntegrityViolationException violationOf(String constraintName) {
    return new DataIntegrityViolationException("could not execute statement",
      new ConstraintViolationException("violation", new SQLException(), constraintName));
  }
}

/**
 * End-to-end: real HTTP on a random port, the whole application, PostgreSQL in Testcontainers.
 * The client is built from the shared TrackLoginApi, exactly like in login-processing-service,
 * so the test also proves that client and server agree on the contract.
 */
@SpringBootTest(webEnvironment = SpringBootTest.WebEnvironment.RANDOM_PORT)
@Testcontainers
class TrackLoginServiceEndToEndIT {

  @Container
  @ServiceConnection
  static PostgreSQLContainer postgres = new PostgreSQLContainer("postgres:18-alpine");

  private static final String TOKEN = "test-token";

  @LocalServerPort
  private int port;

  @Autowired
  private JdbcTemplate jdbc;

  private TrackLoginApi api;

  // Only the signature check against the authorization server's keys is replaced (no Keycloak in tests).
  // Everything around it is real: the filter reads the Authorization header, the authentication is built
  // from the claims ("scope" -> SCOPE_ authorities), and SecurityConfig decides.
  @MockitoBean
  private JwtDecoder jwtDecoder;

  @BeforeEach
  void setUp() {
    jdbc.update("delete from login_event");
    when(jwtDecoder.decode(TOKEN)).thenReturn(Jwt.withTokenValue(TOKEN)
      .header("alg", "none")
      .subject("login-processing-service")
      .claim("scope", "logins:write")
      .build());
    // As login-processing-service will call: every request with its bearer token.
    api = authenticatedClient();
  }

  @Test
  void storesEvent() {
    api.trackLogin(loginEvent());

    assertThat(storedRows()).isEqualTo(1);
  }

  @Test
  void acceptsDuplicateWithoutStoringItAgain() {
    api.trackLogin(loginEvent());

    // Same messageId again (redelivery, retry): still 200 - no exception - and still one row with the original data.
    api.trackLogin(loginEvent("10.0.0.99"));

    assertThat(storedRows()).isEqualTo(1);
    assertThat(jdbc.queryForObject("select customer_ip from login_event", String.class)).isEqualTo(CUSTOMER_IP);
  }

  @Test
  void rejectsInvalidEventWithProblemDetailsAndWithoutStoringIt() {
    HttpClientErrorException.BadRequest exception = catchThrowableOfType(
      HttpClientErrorException.BadRequest.class, () -> api.trackLogin(loginEvent("999.1.1.1")));

    // The client receives the standard error format...
    assertThat(Objects.requireNonNull(exception.getResponseHeaders()).getContentType())
      .isEqualTo(MediaType.APPLICATION_PROBLEM_JSON);
    // ...with the invalid field named, so the caller knows what to fix.
    String body = exception.getResponseBodyAsString();
    List<String> fields = JsonPath.read(body, "$.errors[*].field");
    assertThat(fields).containsExactly("customerIp");

    assertThat(storedRows()).isZero();
  }

  @Test
  void rejectsRequestWithoutToken() {
    TrackLoginApi anonymous = anonymousClient();

    // No Authorization header at all: the real filter chain on the real server answers 401 before the controller.
    assertThatThrownBy(() -> anonymous.trackLogin(loginEvent()))
      .isInstanceOf(HttpClientErrorException.Unauthorized.class);

    assertThat(storedRows()).isZero();
  }

  /** The Prometheus scrape has no token: the endpoint must be open, unlike the API. */
  @Test
  void exposesPrometheusMetricsWithoutToken() {
    String metrics = RestClient.create("http://localhost:" + port)
      .get().uri("/actuator/prometheus")
      .retrieve()
      .body(String.class);

    assertThat(metrics).contains("jvm_memory_used_bytes");
  }

  private int storedRows() {
    return jdbc.queryForObject("select count(*) from login_event", Integer.class);
  }

  // As login-processing-service calls: every request carries its bearer token.
  private TrackLoginApi authenticatedClient() {
    RestClient restClient = RestClient.builder()
      .baseUrl("http://localhost:" + port)
      .defaultHeader(HttpHeaders.AUTHORIZATION, "Bearer " + TOKEN)
      .build();
    return HttpServiceProxyFactory.builderFor(RestClientAdapter.create(restClient)).build()
      .createClient(TrackLoginApi.class);
  }

  // A caller without any token.
  private TrackLoginApi anonymousClient() {
    RestClient restClient = RestClient.create("http://localhost:" + port);
    return HttpServiceProxyFactory.builderFor(RestClientAdapter.create(restClient)).build()
      .createClient(TrackLoginApi.class);
  }
}
